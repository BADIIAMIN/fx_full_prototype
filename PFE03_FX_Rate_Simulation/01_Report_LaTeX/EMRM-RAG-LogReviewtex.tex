	%=============================================================================
	\section{Model Description and Calibration}
	%=============================================================================
	
	\begin{frame}{Model Overview}
		\begin{block}{Model Purpose}
			Simulate FX rate evolution for Potential Future Exposure (PFE) 
		\end{block}
		
		\vspace{0.3cm}
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{Model Framework:}
				\begin{itemize}
					\item Geometric Brownian Motion (GBM)
					\item Log-normal distribution
					\item Constant volatility
					\item Independent currency pairs
				\end{itemize}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\textbf{Key Features:}
				\begin{itemize}
					\item Risk-neutral drift (CIP)
					\item Historical volatility calibration
					\item 3-year lookback window
					\item 5-day return methodology
				\end{itemize}
			\end{column}
		\end{columns}
	\end{frame}
	
	\begin{frame}{Model Dynamics}
		\begin{block}{Stochastic Process}
			Each FX rate $Y_k^{FX}(t)$ follows:
			\begin{equation*}
				dX_k^{FX}(t) = \sigma_k\,dW_k(t)
			\end{equation*}
			\begin{equation*}
				Y_k^{FX}(t) = g_k^{FX}(t)\exp\!\left(X_k^{FX}(t) - \tfrac{1}{2}\sigma_k^2 t\right)
			\end{equation*}
		\end{block}
		
		\vspace{0.2cm}
		
		where:
		\begin{itemize}
			\item $W_k(t)$: Standard Brownian motion
			\item $\sigma_k$: Constant volatility parameter
			\item $g_k^{FX}(t) = Y_k^{FX}(0)\frac{DF_f(0,t)}{DF_d(0,t)}$: Mean function (covered interest parity)
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Calibration Methodology}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{Drift Calibration (Risk-Neutral $\mathbb{Q}$):}
				\begin{equation*}
					g_k^{FX}(t) = Y_k^{FX}(0)\frac{DF_f(0,t)}{DF_d(0,t)}
				\end{equation*}
				
				\begin{itemize}
					\item Derived from interest rate curves
					\item Ensures no-arbitrage at $t=0$
					\item Consistent with forward prices
				\end{itemize}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\textbf{Volatility Calibration (Historical $\mathbb{P}$):}
				\begin{equation*}
					\sigma_k = \text{StdDev}\left[\ln\!\left(\frac{Y_k(t+5d)}{Y_k(t)}\right)\right] \times \frac{1}{\sqrt{5}}
				\end{equation*}
				
				\begin{itemize}
					\item 3-year historical window
					\item 5-day log returns
					\item Scaled to daily frequency
				\end{itemize}
			\end{column}
		\end{columns}
		
		\vspace{0.3cm}
		
		\begin{alertblock}{Mixed-Measure Approach}
			\centering
			Drift from $\mathbb{Q}$ + Volatility from $\mathbb{P}$ = \textbf{Theoretical Inconsistency}
		\end{alertblock}
	\end{frame}
	
	\begin{frame}{Calibration Inputs}
		\begin{block}{Required Data}
			For each FX pair $k$:
		\end{block}
		
		\begin{enumerate}
			\item \textbf{Historical Data:} 3-year time series of daily FX spot rates
			\begin{itemize}
				\item Used to compute 5-day log returns
				\item Calculate standard deviation
			\end{itemize}
			
			\item \textbf{Interest Rate Curves:} Current discount factors
			\begin{itemize}
				\item Domestic curve: $DF_d(0,t)$
				\item Foreign curve: $DF_f(0,t)$
			\end{itemize}
			
			\item \textbf{Current Spot:} $Y_k^{FX}(0)$
			\begin{itemize}
				\item Market observable at calibration time
			\end{itemize}
		\end{enumerate}
	\end{frame}

	\section{Model Assumptions}
	%=============================================================================
	
	\begin{frame}{Core Assumptions (1/3)}
		\begin{block}{Assumption 1: Mixed-Measure Calibration}
			\textbf{Drift:} Risk-neutral (from CIP) \\
			\textbf{Volatility:} Historical (from realized returns)
			
			\textcolor{mufgred}{$\Rightarrow$ Assumes $\sigma^{\mathbb{P}} \approx \sigma^{\mathbb{Q}}$ (ignores vol risk premium)}
		\end{block}
		
		\vspace{0.2cm}
		
		\begin{block}{Assumption 2: Log-Normal Distribution}
			FX rates follow log-normal distribution implying:
			\begin{itemize}
				\item Rates strictly positive
				\item Log-returns normally distributed
				\item No fat tails (thin-tailed)
			\end{itemize}
			\textcolor{mufgred}{$\Rightarrow$ Underestimates extreme events}
		\end{block}
	\end{frame}
	
	\begin{frame}{Core Assumptions (2/3)}
		\begin{block}{Assumption 3: Constant Volatility}
			Volatility $\sigma_k$ is time-invariant and state-independent
			\begin{equation*}
				\text{Var}[r_{0,t}] = \sigma_k^2 t \quad \text{(linear in time)}
			\end{equation*}
			\textcolor{mufgred}{$\Rightarrow$ Ignores volatility clustering, term structure, stochastic volatility}
		\end{block}
		
		\vspace{0.2cm}
		
		\begin{block}{Assumption 4: Historical Calibration}
			\begin{itemize}
				\item 5-day returns assumed i.i.d.
				\item 3-year window stationary
				\item Variance scales linearly: $\sigma_{\text{daily}} = \sigma_{5d}/\sqrt{5}$
			\end{itemize}
			\textcolor{mufgred}{$\Rightarrow$ smooths spikes, ignores regime changes}
		\end{block}
	\end{frame}
	
	\begin{frame}{Core Assumptions (3/3)}


		\begin{block}{Assumption 7: No Jumps}
			Pure diffusion, no discontinuous movements
			
			\textcolor{mufgred}{$\Rightarrow$ Misses central bank interventions, crashes, news shocks}
		\end{block}
	\end{frame}
	
	\begin{frame}{Assumptions Summary Table}
		\scriptsize
		\begin{table}
			\begin{tabular}{|p{3.5cm}|p{4cm}|p{3.5cm}|}
				\hline
				\textbf{Assumption} & \textbf{Model Specification} & \textbf{Reality} \\
				\hline
				Mixed-measure & $\mathbb{Q}$ drift + $\mathbb{P}$ vol & Vol risk premium: $\sigma^{\mathbb{Q}} > \sigma^{\mathbb{P}}$ \\
				\hline
				Log-normal & Normal log-returns & Fat tails, kurtosis $\approx$ 5--10 \\
				\hline
				Constant volatility & $\sigma(t,S) = \sigma$ & Clustering, smile, term structure \\
				\hline
				Historical calibration & 5-day, 3-year window & Market uses implied vol \\
				\hline
				Independence & $\rho_{ij} = 0$ & Empirical: $\rho \approx$ 0.65 (USD crosses) \\
				\hline
				Deterministic rates & $DF(0,t)$ frozen & Rates stochastic, correlated with FX \\
				\hline
				No jumps & Pure diffusion & Central bank interventions, crises \\
				\hline
			\end{tabular}
		\end{table}
	\end{frame}

	\section{Model Limitations}
	%=============================================================================
	
	\begin{frame}{Limitation 1: Mixed-Measure Calibration}
		\begin{alertblock}{Fundamental Inconsistency}
			Model combines risk-neutral drift with historical volatility
		\end{alertblock}
		
		\vspace{0.2cm}
		
		\textbf{Theoretical Issues:}
		\begin{itemize}
			\item \textbf{Volatility Risk Premium Ignored:} Market implied vol exceeds realized vol by 2--4\%
			\item \textbf{FX Risk Premium Omitted:} UIP fails empirically (carry trades profitable)
			\item \textbf{Measure Ambiguity:} Neither pure $\mathbb{Q}$ nor pure $\mathbb{P}$
		\end{itemize}
		
		\vspace{0.2cm}
		
		\textbf{Impact:}
		\begin{itemize}
			\item[$\times$] PFE  underestimated by \textcolor{mufgred}{\textbf{8--20\%}}
			\item[$\times$] Option values underpriced by 10--15\%
			\item[$\times$] Hedge ratios wrong by 15--30\%
		\end{itemize}
	\end{frame}
	
		\begin{frame}{Limitation 2: Deterministic Interest Rates}
		\begin{block}{Pathwise No-Arbitrage Failure}
			Frozen discount factors create arbitrage opportunities on individual paths
		\end{block}
		
		\vspace{0.2cm}
		
		\textbf{Problem:} Conditional expectation incorrect
		\begin{align*}
			\text{Should be:} \quad & \mathbb{E}_s^{\mathbb{Q}}[S(t)] = S(s) \frac{DF_f(s,t)}{DF_d(s,t)} \\
			\text{Model uses:} \quad & \mathbb{E}_s^{\mathbb{Q}}[S(t)] = S(0) \frac{DF_f(0,t)}{DF_d(0,t)} \quad \textcolor{mufgred}{\leftarrow \text{Wrong!}}
		\end{align*}
		
		\vspace{0.2cm}
		
		\textbf{Impact:}
		\begin{itemize}

			\item Long-dated PFE error: \textbf{15--30\%}

			\item Callable/Bermudan mispricing
		\end{itemize}
	\end{frame}
	
	
	\begin{frame}{Limitation 3: Constant Volatility}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{What the Model Misses:}
				\begin{enumerate}
					\item \textbf{Volatility Clustering}
					\begin{itemize}
						\item High vol $\rightarrow$ high vol
						\item Low vol $\rightarrow$ low vol
						\item Persistence: $\alpha+\beta \approx 0.95$
					\end{itemize}
					
					\item \textbf{Term Structure}
					\begin{itemize}
						\item Short-term vol $>$ long-term
						\item Mean-reversion effects
					\end{itemize}
					
					\item \textbf{Stochastic Volatility}
					\begin{itemize}
						\item Vol itself is random
						\item Creates smile/skew
					\end{itemize}
				\end{enumerate}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\textbf{Quantitative Impact:}
				\begin{itemize}
					\item PFE underestimation: \textcolor{mufgred}{\textbf{15--30\%}}
					\item Tail risk (VaR$_{99\%}$): \textbf{20--40\%} error
					\item Path-dependent mispricing 
				\end{itemize}
				
				\vspace{0.3cm}
				
				\textbf{Example:}
				\begin{itemize}
					\item Calibrated: $\sigma = 8\%$
					\item Stress spike: $\sigma = 14\%$
					\item PFE error: \textbf{38\%} underestimation
				\end{itemize}
			\end{column}
		\end{columns}
	\end{frame}

	\section{Statistical Validation}
	%=============================================================================
	
	\begin{frame}{Test 1: Ljung-Box (Volatility Clustering)}
		\textbf{Purpose:} Test for autocorrelation in squared returns (volatility clustering)
		
		\vspace{0.2cm}
		
		\textbf{Null Hypothesis:} $H_0$: $\text{Corr}(r_t^2, r_{t-k}^2) = 0$ for all $k$
		
		\vspace{0.2cm}
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{Test Statistic:}
				\begin{equation*}
					Q(K) = T(T+2) \sum_{k=1}^K \frac{\hat{\rho}_k^2}{T-k} \sim \chi^2(K)
				\end{equation*}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\textbf{Results (EUR/USD):}
				\scriptsize
				\begin{table}
					\begin{tabular}{|c|c|c|}
						\hline
						\textbf{Lag} & \textbf{Q-stat} & \textbf{p-value} \\
						\hline
						1 & 24.8 & $<0.001$ \\
						5 & 87.3 & $<0.001$ \\
						10 & 142.5 & $<0.001$ \\
						20 & 198.2 & $<0.001$ \\
						\hline
					\end{tabular}
				\end{table}
				\normalsize
			\end{column}
		\end{columns}
		
		\vspace{0.3cm}
		
		\begin{alertblock}{Conclusion}
			\textcolor{mufgred}{\textbf{REJECT $H_0$}} — Strong evidence of volatility clustering. Constant volatility assumption \textbf{VIOLATED}.
		\end{alertblock}
	\end{frame}
	
	\begin{frame}{Test 2: ARCH-LM (Heteroskedasticity)}
		\textbf{Purpose:} Test specifically for ARCH effects (volatility clustering)
		
		\vspace{0.2cm}
		
		\textbf{Null Hypothesis:} $H_0$: No ARCH effects (constant volatility)
		
		\vspace{0.2cm}
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\textbf{Procedure:}
				\begin{enumerate}
					\item Regress $r_t$ on lags
					\item Get residuals $\hat{\epsilon}_t$
					\item Regress $\hat{\epsilon}_t^2$ on lags
					\item Compute LM statistic
				\end{enumerate}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\textbf{Results (EUR/USD):}
				\begin{itemize}
					\item LM Statistic: 127.4
					\item p-value: $< 0.001$
					\item F-statistic: 13.2
					\item F p-value: $< 0.001$
				\end{itemize}
			\end{column}
		\end{columns}
		
		\vspace{0.3cm}
		
		\begin{alertblock}{Conclusion}
			\textcolor{mufgred}{\textbf{REJECT $H_0$}} — Overwhelming evidence of ARCH effects. Constant volatility model \textbf{INADEQUATE}.
		\end{alertblock}
	\end{frame}

	%=============================================================================
	\section{Conclusions and Recommendations}
	%=============================================================================
	
	\begin{frame}{Key Findings}
		\begin{enumerate}
			\item \textbf{Fundamental Issue: Mixed-Measure Calibration}
			\begin{itemize}
				\item Combines $\mathbb{Q}$ drift with $\mathbb{P}$ volatility
				\item Systematic underestimation: \textcolor{mufgred}{\textbf{8--20\%}}
				\item Affects all risk metrics
			\end{itemize}
			
			\item \textbf{Constant Volatility Fails Empirically}
			\begin{itemize}
				\item All statistical tests reject assumption
				\item Misses clustering, term structure, stochastic dynamics

			\end{itemize}
			
			\item \textbf{Missing Features Compound Errors}
			\begin{itemize}
				\item No correlation: portfolio error
				\item No jumps:  tail risk error
				\item Deterministic rates: XCS pricing error
				\item Path-dependent:  mispricing
			\end{itemize}
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Material Impact by Portfolio Segment}
		\begin{block}{High Impact Segments}
			\begin{itemize}
				\item \textcolor{mufgred}{\textbf{Long-dated exposures}} ($>2$ years): Deterministic rates + constant vol
				\item \textcolor{mufgred}{\textbf{Barrier-heavy portfolios}}: Jump risk + smile omission
				\item \textcolor{mufgred}{\textbf{Cross-currency swaps}}: Pathwise no-arbitrage + rate-FX correlation
				\item \textcolor{mufgred}{\textbf{Multi-currency portfolios}}: Correlation structure critical
			\end{itemize}
		\end{block}
		
		\vspace{0.3cm}
		
		\begin{block}{Acceptable for}
			\begin{itemize}
				\item Short-dated ($<6$M) vanilla forwards
				\item Single-currency, linear exposures
				\item Low-volatility major pairs (EUR/USD, USD/JPY) in calm markets
			\end{itemize}
		\end{block}
	\end{frame}
	
	\begin{frame}{Recommendations}
		\begin{block}{Priority/Option 1}
			\begin{enumerate}
				\item \textbf{Volatility Adjustment:} Apply 10--15\% uplift to historical vol
				\begin{equation*}
					\sigma_{\text{adjusted}} = \sigma_{\text{historical}} \times 1.12

	\begin{frame}[allowframebreaks]{Backup: Technical Details}
		\frametitle{Mathematical Formulation}
		
		\textbf{SDE Specification:}
		\begin{align*}
			dX_k^{FX}(t) &= \sigma_k dW_k(t) \\
			Y_k^{FX}(t) &= g_k^{FX}(t) \exp\left(X_k^{FX}(t) - \frac{1}{2}\sigma_k^2 t\right) \\
			g_k^{FX}(t) &= Y_k^{FX}(0) \frac{DF_f(0,t)}{DF_d(0,t)}
		\end{align*}
		
		\framebreak
		
		\textbf{Log-Return Distribution:}
		\begin{equation*}
			R_{s,t} := \ln\left(\frac{Y_k^{FX}(t)}{Y_k^{FX}(s)}\right) \sim \mathcal{N}\left((r_d - r_f - \frac{1}{2}\sigma_k^2)(t-s), \sigma_k^2(t-s)\right)
		\end{equation*}
		
		\framebreak
		
		\textbf{Monte Carlo Implementation:}
		\begin{enumerate}
			\item Generate $Z \sim \mathcal{N}(0,1)$
			\item Update: $X(t+\Delta t) = X(t) + \sigma\sqrt{\Delta t} \cdot Z$
			\item Compute: $Y(t+\Delta t) = g(t+\Delta t) \exp(X(t+\Delta t) - \frac{1}{2}\sigma^2(t+\Delta t))$
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Backup: GARCH Specification}
		\textbf{GARCH(1,1) Model:}
		\begin{align*}
			r_t &= \mu + \sigma_t \epsilon_t, \quad \epsilon_t \sim \mathcal{N}(0,1) \\
			\sigma_t^2 &= \omega + \alpha r_{t-1}^2 + \beta \sigma_{t-1}^2
		\end{align*}
		
		\vspace{0.3cm}
		
		\textbf{Interpretation:}
		\begin{itemize}
			\item $\omega$: Long-run baseline variance
			\item $\alpha$: Reaction to recent shocks (ARCH effect)
			\item $\beta$: Persistence of volatility (GARCH effect)
			\item $\alpha + \beta$: Overall persistence (near 1 = long memory)
		\end{itemize}
		
		\vspace{0.3cm}
		
		\textbf{Empirical Values (EUR/USD):}
		\begin{itemize}
			\item $\omega = 0.0008$, $\alpha = 0.082$, $\beta = 0.905$
			\item Persistence: $\alpha + \beta = 0.987$ (very close to 1)
			\item Half-life: $\approx 53$ days
		\end{itemize}
	\end{frame}
